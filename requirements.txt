# =============================================================================
# Smart Gate Attendance - Python 3.10 / 3.11 / 3.12
#
#   pip install -r requirements.txt
#
# One file for every target; environment markers pick the right packages:
#   Windows / Linux x86_64 (PC, H100 / A100 / L4 cloud GPU) -> onnxruntime-gpu + CUDA 12 / cuDNN 9 as pip wheels
#                                                             (only the NVIDIA driver is needed on the host)
#   NVIDIA Jetson (aarch64, JetPack 6)                     -> see the Jetson section at the bottom
# Tested: onnxruntime-gpu 1.22.0, opencv 4.11, numpy 2.4, starlette 1.7, uvicorn 0.54 (RTX 4060, CUDA 12)
# =============================================================================

# ---- inference: ONNX Runtime GPU (x86_64) ----
onnxruntime-gpu[cuda,cudnn]>=1.22,<1.23 ; platform_machine == "x86_64" or platform_machine == "AMD64"

# ---- vision ----
# GUI build on Windows (OpenCV test window); headless on Linux servers (no X11 libs needed).
# Jetson: use JetPack's OpenCV (GStreamer + hardware decode) instead of a pip wheel.
opencv-python>=4.8,<5 ; sys_platform == "win32"
opencv-python-headless>=4.8,<5 ; sys_platform == "linux" and platform_machine == "x86_64"
numpy>=1.24,<3 ; platform_machine != "aarch64"
numpy>=1.24,<2 ; platform_machine == "aarch64"   # Jetson ORT wheels are built against NumPy 1.x
scipy>=1.10,<2                                   # Hungarian matching in the face tracker

# ---- app ----
PyYAML>=6.0,<7
loguru>=0.7,<1
openpyxl>=3.1,<4
starlette>=0.37,<2
uvicorn>=0.29,<1
python-multipart>=0.0.9                          # photo upload on the dashboard

# ---- one-time FP16 conversion of ArcFace for backend "cuda" (not needed for "tensorrt") ----
onnx>=1.15,<2
onnxconverter-common>=1.14,<2

# =============================================================================
# OPTIONAL - TensorRT FP16 (backend: "tensorrt", fastest on H100 / A100 / L4 / RTX).
# The TensorRT major version must match onnxruntime-gpu: 1.22 is built for TensorRT 10.x
# (TensorRT 11 will NOT load and the app falls back to CUDA). Uncomment on x86_64:
# tensorrt-cu12==10.9.*
# =============================================================================
#
# NVIDIA JETSON (Orin Nano / NX / AGX, JetPack 6):
#   1. sudo apt install python3-opencv            (or JetPack's OpenCV with GStreamer)
#   2. pip install onnxruntime-gpu from NVIDIA's Jetson wheel index for your JetPack version
#      (TensorRT comes with JetPack; set inference.backend: "tensorrt")
#   3. pip install -r requirements.txt            (markers skip the x86-only packages)
#
# CPU-ONLY (no NVIDIA GPU), set inference.backend: "cpu" - expect a few FPS per camera:
#   x86_64 : replace the onnxruntime-gpu line above with  onnxruntime>=1.22,<1.23  (saves ~2 GB of CUDA wheels)
#   ARM64 (Raspberry Pi etc.) : pip install onnxruntime opencv-python-headless, then this file
